results=[]
for name,b in [('production',core),('conventional',sql),('floor',Backend('floor'))]:
 s=b.start(world(),{'records':{'RET1':{'received':True}}});o=Oracle(s);o.clock(1);o.reconcile(s)

 actual={d:o.board()[d]['resolution'] for d in ['D_refund(CL1,120)','D_resolve(CL1)']}
 expected={d:'discharged' for d in actual}
 results.append(dict(backend=name,expected=expected,actual=actual,passed=actual==expected,
