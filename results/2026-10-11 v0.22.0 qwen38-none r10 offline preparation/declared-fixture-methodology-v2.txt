"""Declared fixtures only: no model calls, credentials, or network traffic."""
import hashlib
import json
import shutil
import sys
import tempfile
from collections import Counter
from pathlib import Path
from unittest.mock import patch

ROOT = Path('/workspace/taa-lab')
sys.path.insert(0, str(ROOT))
from formation.evaluation_runner_v0218 import ADAPTERS, run_batch

from formation.limits import RatePaused

from scripts.formation_study_a_gate3 import verify_freeze

fixture_paths = ['studies/formation-study-a-fixtures-v0213.json',
                 'studies/formation-study-a-pressure-fixtures-v0215.json']
fixtures = []
for path in fixture_paths:
    fixtures += json.loads((ROOT / path).read_text())['fixtures']
packets = []
for cell in cells:
    packets += next(f['packets'] for f in fixtures
                    if f['case_id'] == cell['case_id'] and f['name'] == 'required_conduct')
requests = []
clock = [time.time()]
def slots(path):
    return GroqSlots(path, now=lambda: clock[0])
with tempfile.TemporaryDirectory() as temporary:
    folder = Path(temporary)
    root = folder / 'isolated-root'
    output = folder / 'DECLARED-FIXTURE-OUTPUT'
    files = set(ADAPTERS) | {PLAN['price_file'], 'studies/formation-study-a-evaluation-plan-v0217.json',
                           'studies/formation-study-a-freeze-v0217.json',
                           'results/formation-spending-v021/groq.json',
                           'results/formation-spending-v021/groq-rate.json'} | {c['path'] for c in cells}
    for name in files:
        target = root / name
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(ROOT / name, target)
    raws = iter(p['raw'] for p in packets)

    def send(request, **kwargs):
        assert request.full_url == 'https://api.groq.com/openai/v1/chat/completions'
        assert request.get_method() == 'POST'
        assert request.get_header('User-agent') == 'TAA-Lab/0.21.2 (Python standard-library API client)'
        body = json.loads(request.data)
        assert body['model'] == MODEL and body['max_completion_tokens'] == 2048
        assert body['messages'][0] == {'role': 'system', 'content': PLAN['system']}
        assert set(body) == {'model', 'messages', 'max_completion_tokens'}
        assert len(request.data) <= 32000
        if requests:
            clock[0] += 61
        requests.append(body)
        return Response(MODEL, 'groq', next(raws))

    with patch('formation.evaluation_runner_v0218.load_keys') as keys, \
         patch.dict('os.environ', {'GROQ_API_KEY': 'declared-fixture-only'}), \
         patch('urllib.request.urlopen', side_effect=send) as transport, \
         patch('formation.evaluation_runner_v0218.GroqSlots', side_effect=slots):
        try:
            run_batch(MODEL, 0.0, output, root=root)
        except ValueError:
            pass
        else:
            raise AssertionError('Unapproved batch was accepted')
        keys.assert_not_called()
        transport.assert_not_called()
        assert not output.exists()
        paused = run_batch(MODEL, 0.0, output, approved=True, confirm_groq_free=True, root=root)
        assert paused['status'] == 'paused' and paused['completed_cells'] == 0
        assert transport.call_count == 1
        state = json.loads((output / 'A_R01-r01-checkpoint.json').read_text())
        assert state['phase'] == 'paused' and len(state['packets']) == 1
        assert not (root / 'results/formation-spending-v021/run.lock').exists()
        clock[0] += 61
        summary = run_batch(MODEL, 0.0, output, approved=True, confirm_groq_free=True, root=root)
        assert summary['status'] == 'complete' and summary['completed_cells'] == 240
        assert summary['planned_cells'] == 240
        assert summary['records'] == [dict(c, outcome='correct', attempted=True,
                                            identity_valid=True, status='complete') for c in cells]
        assert transport.call_count == len(packets) == len(requests)
        calls_before_resume = transport.call_count
        isolated_ledger_before_resume = (root / 'results/formation-spending-v021/groq.json').read_bytes()
        resumed = run_batch(MODEL, 0.0, output, approved=True, confirm_groq_free=True, root=root)
        assert resumed['completed_cells'] == 240
        assert transport.call_count == calls_before_resume
        assert isolated_ledger_before_resume == (root / 'results/formation-spending-v021/groq.json').read_bytes()
        try:
            run_batch(MODEL, 1.0, output, approved=True, confirm_groq_free=True, root=root)
        except ValueError:
            pass
        else:
            raise AssertionError('Resume cap expansion was accepted')
        assert transport.call_count == calls_before_resume
        keys.assert_called()
    assert not (root / 'results/formation-spending-v021/run.lock').exists()
    manifest = json.loads((output / 'execution-manifest.json').read_text())
    assert manifest['budget'] == 0.0 and manifest['cells'] == cells
    prior = json.loads(ledger.read_text())['estimated_api_dollars']
    journal = [json.loads(line) for line in (output / 'calls.jsonl').read_text().splitlines()]
    starts = [e for e in journal if e['event'] == 'attempt_start']
    ends = [e for e in journal if e['event'] == 'attempt_end']
    assert len(starts) == len(ends) == len(packets)
    assert all(e['usage']['completion_tokens'] == 50 for e in ends)
    assert all(e['model_returned'] == MODEL and not e['error'] and not e['usage_missing'] for e in ends)
    assert round(json.loads((root / 'results/formation-spending-v021/groq.json').read_text())['estimated_api_dollars']
                 - prior, 8) == round(summary['batch_estimated_dollars'], 8)
    result = {
        'phase': 'DECLARED NATIVE-RESPONSE FIXTURE; not an experimental model observation',
        'model_transport': MODEL, 'fixture_episodes': 240, 'passed': 240,
        'repeats': 10, 'arms': dict(Counter(c['arm'] for c in cells)),
        'mocked_requests': len(packets), 'actual_network_requests': 0, 'model_calls': 0,
        'actual_api_cost_dollars': 0, 'mocked_accounting_dollars': summary['batch_estimated_dollars'],
        'proposed_batch_cap_dollars': 0.0, 'prior_cumulative_ledger_dollars': prior,
        'quota_pause_then_same_output_resume_passed': True, 'simulated_clock_only': True,
        'approval_barrier_passed': True, 'resume_without_duplicate_requests_passed': True,
        'resume_cap_expansion_rejected': True, 'native_wire_defaults_and_pinned_identity_passed': True,
        'max_wire_bytes': max(len(json.dumps(b).encode()) for b in requests),
        'frozen_files_verified': verify_freeze(),
        'real_provider_ledger_sha256_before': before_ledger,
        'fixture_input_sha256': {p: hashlib.sha256((ROOT / p).read_bytes()).hexdigest() for p in fixture_paths},

        'declared_records': summary['records']}
rate = ROOT / 'results/formation-spending-v021/groq-rate.json'
