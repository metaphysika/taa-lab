def read(path):
    return json.loads((ROOT / path).read_text())

def sha(path):
    return hashlib.sha256((ROOT / path).read_bytes()).hexdigest()

def write(path, data):
    with (ROOT / path).open('x') as out:
        json.dump(data, out, indent=2)
        out.write('\n')
